  2. defLax48.TwinWidth p. 1

    Twin-width

    A partition sequence of a finite simple graph G is a sequence of partitions of its vertex set that starts at the partition into singletons, merges two parts into one at every step, and ends at the partition with a single part. Two parts of a same partition are homogeneous if either every pair of vertices across them is adjacent or none is; two non-homogeneous parts are red-adjacent. The red degree of a part is the number of other parts of its partition that are red-adjacent to it.

    The twin-width of G is the least d such that G has a partition sequence in which every part of every partition has red degree at most d.

    33/-- Two vertex sets are homogeneous in `G`: either every pair of vertices
    34across them is adjacent, or none is. -/
    35def Homogeneous {V : Type} (G : SimpleGraph V) (A B : Finset V) : Prop :=
    36 (∀ a ∈ A, ∀ b ∈ B, G.Adj a b) ∨ (∀ a ∈ A, ∀ b ∈ B, ¬ G.Adj a b)
    37
    38/-- The red degree of a part `A` in a family of parts `P`: the number of
    39other parts of `P` that are not homogeneous with `A`. -/
    40noncomputable def redDegree {V : Type} (G : SimpleGraph V)
    41 (P : Finset (Finset V)) (A : Finset V) : ℕ :=
    42 {B | B ∈ P ∧ B ≠ A ∧ ¬ Homogeneous G A B}.ncard
    43
    44/-- The partition of a finite vertex type into singletons. -/
    45def singletonPartition (V : Type) [Fintype V] [DecidableEq V] :
    46 Finset (Finset V) :=
    47 Finset.univ.image fun v : V => ({v} : Finset V)
    48
    49/-- A partition sequence of `G` in which every part has red degree at most
    50`d`: starting from the singleton partition, each step merges two parts into
    51one, until a single part remains. -/
    52structure PartitionSequence {V : Type} [Fintype V] [DecidableEq V]
    53 (G : SimpleGraph V) (d : ℕ) where
    54 /-- The number of merge steps. -/
    55 stepCount : ℕ
    56 /-- The partition after each number of merge steps. -/
    57 partition : ℕ → Finset (Finset V)
    58 /-- The sequence starts at the singleton partition. -/
    59 starts : partition 0 = singletonPartition V
    60 /-- The sequence ends with a single part. -/
    61 ends : (partition stepCount).card ≤ 1
    62 /-- Each step merges two distinct parts into one and keeps all other
    63 parts. -/
    64 step_merges :
    65 ∀ i, i < stepCount → ∃ A ∈ partition i, ∃ B ∈ partition i, A ≠ B ∧
    66 partition (i + 1) = insert (A ∪ B) (((partition i).erase A).erase B)
    67 /-- Every part of every partition in the sequence has red degree at most
    68 `d`. -/
    69 redDegree_le :
    70 ∀ i, i ≤ stepCount → ∀ ⦃A⦄, A ∈ partition i →
    71 redDegree G (partition i) A ≤ d
    72
    73/-- `G` has a partition sequence in which every part has red degree at most
    74`d`. -/
    75def HasTwinWidthAtMost {V : Type} [Fintype V] [DecidableEq V]
    76 (G : SimpleGraph V) (d : ℕ) : Prop :=
    77 Nonempty (PartitionSequence G d)
    78
    79/-- The twin-width of a finite simple graph: the least `d` such that the
    80graph has a partition sequence in which every part has red degree at most
    81`d`. -/
    82noncomputable def twinWidth {V : Type} [Fintype V] [DecidableEq V]
    83 (G : SimpleGraph V) : ℕ :=
    84 sInf {d | HasTwinWidthAtMost G d}

  3. defLax48.Treewidth p. 2

    Treewidth

    A tree decomposition of a finite simple graph G consists of a finite tree T and a bag of graph vertices at each node such that every graph vertex occurs in a bag, the endpoints of every graph edge occur together in a bag, and the nodes whose bags contain any fixed vertex induce a connected subgraph of T.

    The treewidth of G is the least w such that G has a tree decomposition with every bag of size at most w + 1.

    28/-- A tree decomposition of a simple graph: a finite tree of nodes with a bag
    29of graph vertices at each node, such that the bags cover every vertex and
    30every edge, and the nodes whose bags contain any fixed vertex induce a
    31connected subgraph of the tree. -/
    32structure TreeDecomposition {V : Type} (G : SimpleGraph V) where
    33 /-- The node type of the decomposition tree. -/
    34 Node : Type
    35 /-- The decomposition tree is finite. -/
    36 [nodeFintype : Fintype Node]
    37 /-- The graph on the decomposition nodes. -/
    38 tree : SimpleGraph Node
    39 /-- The node graph is a tree. -/
    40 isTree : tree.IsTree
    41 /-- The bag assigned to each decomposition node. -/
    42 bag : Node → Finset V
    43 /-- Every graph vertex appears in at least one bag. -/
    44 vertex_mem_bag : ∀ v : V, ∃ i : Node, v ∈ bag i
    45 /-- Every graph edge has both endpoints together in at least one bag. -/
    46 edge_mem_bag : ∀ ⦃u v : V⦄, G.Adj u v → ∃ i : Node, u ∈ bag i ∧ v ∈ bag i
    47 /-- For each graph vertex, the nodes whose bags contain it induce a
    48 connected subgraph of the tree. -/
    49 bag_indices_connected :
    50 ∀ v : V, (tree.induce {i : Node | v ∈ bag i}).Connected
    51
    52/-- `G` has a tree decomposition all of whose bags have at most `w + 1`
    53vertices. -/
    54def HasTreewidthAtMost {V : Type} (G : SimpleGraph V) (w : ℕ) : Prop :=
    55 ∃ D : TreeDecomposition G, ∀ i, (D.bag i).card ≤ w + 1
    56
    57/-- The treewidth of a finite simple graph: the least `w` such that the
    58graph has a tree decomposition with bags of at most `w + 1` vertices. -/
    59noncomputable def treewidth {V : Type} [Fintype V] [DecidableEq V]
    60 (G : SimpleGraph V) : ℕ :=
    61 sInf {w | HasTreewidthAtMost G w}
    62
    63end Lax48.Treewidth

    Grid number

    A k-division of a matrix partitions its rows into k sets of consecutive rows and its columns into k sets of consecutive columns; its k² cells are the intersections of a row set with a column set. A matrix has a k-grid minor if it has a k-division with at least one 1-entry in each of its k² cells. The grid number of a matrix is the largest k such that it has a k-grid minor. The grid number of a graph G, denoted by gn(G), is the minimum, taken among all the adjacency matrices M of G, of the grid number of M.

    28/-- `M` has a `k`-grid minor: a division of the rows and of the columns into
    29`k` sets of consecutive indices such that every cell contains a 1-entry. -/
    30def HasGridMinor {n : ℕ} (M : Fin n → Fin n → Prop) (k : ℕ) : Prop :=
    31 ∃ r c : Fin (k + 1) → ℕ,
    32 StrictMono r ∧ r 0 = 0 ∧ r (Fin.last k) = n ∧
    33 StrictMono c ∧ c 0 = 0 ∧ c (Fin.last k) = n ∧
    34 ∀ i j : Fin k, ∃ a b : Fin n,
    35 r i.castSucc ≤ a.val ∧ a.val < r i.succ ∧
    36 c j.castSucc ≤ b.val ∧ b.val < c j.succ ∧ M a b
    37
    38/-- The grid number of a 0/1 matrix: the largest `k` such that it has a
    39`k`-grid minor. -/
    40noncomputable def matrixGridNumber {n : ℕ} (M : Fin n → Fin n → Prop) : ℕ :=
    41 sSup {k | HasGridMinor M k}
    42
    43/-- The adjacency matrix of `G` under the vertex ordering `σ`. -/
    44def adjacencyMatrix {V : Type} (G : SimpleGraph V) {n : ℕ} (σ : Fin n ≃ V) :
    45 Fin n → Fin n → Prop :=
    46 fun a b => G.Adj (σ a) (σ b)
    47
    48/-- The grid number of a finite simple graph: the least grid number of one of
    49its adjacency matrices. -/
    50noncomputable def gridNumber {V : Type} [Fintype V] [DecidableEq V]
    51 (G : SimpleGraph V) : ℕ :=
    52 sInf {k | ∃ σ : Fin (Fintype.card V) ≃ V,
    53 matrixGridNumber (adjacencyMatrix G σ) = k}

  5. defLax65.OrientedTwinWidth p. 3

    Oriented twin-width

    An oriented partition sequence of a finite simple graph G is a partition sequence of G in which every red edge is replaced by a red arc leaving the newly merged part: when two parts are merged, the merged part gets a red arc to every other part it is not homogeneous with, and the red arcs between the remaining parts stay as they were. An oriented d-sequence is one in which every part of every partition has at most d out-going red arcs. The oriented twin-width of G is the least d such that G admits an oriented d-sequence.

    28open Lax48.TwinWidth

    30/-- An oriented partition sequence of `G` in which every part has at most `d`
    31out-going red arcs: starting from the singleton partition with no red arcs,
    32each step merges two parts into one, gives the merged part a red arc to every
    33other part it is not homogeneous with, and keeps the red arcs between the
    34remaining parts, until a single part remains. -/
    35structure OrientedPartitionSequence {V : Type} [Fintype V] [DecidableEq V]
    36 (G : SimpleGraph V) (d : ℕ) where
    37 /-- The number of merge steps. -/
    38 stepCount : ℕ
    39 /-- The partition after each number of merge steps. -/
    40 partition : ℕ → Finset (Finset V)
    41 /-- The red arcs between parts after each number of merge steps. -/
    42 redArc : ℕ → Finset V → Finset V → Prop
    43 /-- The sequence starts at the singleton partition. -/
    44 starts : partition 0 = singletonPartition V
    45 /-- There are no red arcs at the start. -/
    46 redArc_zero : ∀ A B, ¬ redArc 0 A B
    47 /-- The sequence ends with a single part. -/
    48 ends : (partition stepCount).card ≤ 1
    49 /-- Each step merges two distinct parts into one and keeps all other parts;
    50 the merged part gets a red arc to every other part it is not homogeneous
    51 with, and the red arcs between the remaining parts are unchanged. -/
    52 step_merges :
    53 ∀ i, i < stepCount → ∃ A ∈ partition i, ∃ B ∈ partition i, A ≠ B ∧
    54 partition (i + 1) = insert (A ∪ B) (((partition i).erase A).erase B) ∧
    55 ∀ C ∈ partition (i + 1), ∀ D ∈ partition (i + 1),
    56 (redArc (i + 1) C D ↔
    57 if C = A ∪ B then D ≠ C ∧ ¬ Homogeneous G C D
    58 else D ≠ A ∪ B ∧ redArc i C D)
    59 /-- Every part of every partition in the sequence has at most `d` out-going
    60 red arcs. -/
    61 outDegree_le :
    62 ∀ i, i ≤ stepCount → ∀ ⦃A⦄, A ∈ partition i →
    63 {B | B ∈ partition i ∧ redArc i A B}.ncard ≤ d
    64
    65/-- `G` admits an oriented `d`-sequence. -/
    66def HasOrientedTwinWidthAtMost {V : Type} [Fintype V] [DecidableEq V]
    67 (G : SimpleGraph V) (d : ℕ) : Prop :=
    68 Nonempty (OrientedPartitionSequence G d)
    69
    70/-- The oriented twin-width of a finite simple graph: the least `d` such that
    71the graph admits an oriented `d`-sequence. -/
    72noncomputable def orientedTwinWidth {V : Type} [Fintype V] [DecidableEq V]
    73 (G : SimpleGraph V) : ℕ :=
    74 sInf {d | HasOrientedTwinWidthAtMost G d}
    75
    76end Lax65.OrientedTwinWidth
    77

    Twin-width can be exponential in treewidth

    For every real 0 < *ε* ≤ 1/2 and integer *t* > 1/ε, there is a graph G<sub>t,ε</sub> with a feedback vertex set of size t and such that tww(G<sub>t,ε</sub>) > 2^{(1−ε)t}.

    The graph G_{t,ε}

    Fix a real ε and an integer t, and set

    Oriented twin-width

    An oriented partition sequence of a finite simple graph G is a partition sequence of G in which every red edge is replaced by a red arc leaving the newly merged part: when two parts are merged, the merged part gets a red arc to every other part it is not homogeneous with, and the red arcs between the remaining parts stay as they were. An oriented d-sequence is one in which every part of every partition has at most d out-going red arcs. The oriented twin-width of G is the least d such that G admits an oriented d-sequence.

    30/-- An oriented partition sequence of `G` in which every part has at most `d`
    31out-going red arcs: starting from the singleton partition with no red arcs,
    32each step merges two parts into one, gives the merged part a red arc to every
    33other part it is not homogeneous with, and keeps the red arcs between the
    34remaining parts, until a single part remains. -/
    35structure OrientedPartitionSequence {V : Type} [Fintype V] [DecidableEq V]
    36 (G : SimpleGraph V) (d : ℕ) where
    37 /-- The number of merge steps. -/
    38 stepCount : ℕ
    39 /-- The partition after each number of merge steps. -/
    40 partition : ℕ → Finset (Finset V)
    41 /-- The red arcs between parts after each number of merge steps. -/
    42 redArc : ℕ → Finset V → Finset V → Prop
    43 /-- The sequence starts at the singleton partition. -/
    44 starts : partition 0 = singletonPartition V
    45 /-- There are no red arcs at the start. -/
    46 redArc_zero : ∀ A B, ¬ redArc 0 A B
    47 /-- The sequence ends with a single part. -/
    48 ends : (partition stepCount).card ≤ 1
    49 /-- Each step merges two distinct parts into one and keeps all other parts;
    50 the merged part gets a red arc to every other part it is not homogeneous
    51 with, and the red arcs between the remaining parts are unchanged. -/
    52 step_merges :
    53 ∀ i, i < stepCount → ∃ A ∈ partition i, ∃ B ∈ partition i, A ≠ B ∧
    54 partition (i + 1) = insert (A ∪ B) (((partition i).erase A).erase B) ∧
    55 ∀ C ∈ partition (i + 1), ∀ D ∈ partition (i + 1),
    56 (redArc (i + 1) C D ↔
    57 if C = A ∪ B then D ≠ C ∧ ¬ Homogeneous G C D
    58 else D ≠ A ∪ B ∧ redArc i C D)
    59 /-- Every part of every partition in the sequence has at most `d` out-going
    60 red arcs. -/
    61 outDegree_le :
    62 ∀ i, i ≤ stepCount → ∀ ⦃A⦄, A ∈ partition i →
    63 {B | B ∈ partition i ∧ redArc i A B}.ncard ≤ d
    64

    66def HasOrientedTwinWidthAtMost {V : Type} [Fintype V] [DecidableEq V]
    67 (G : SimpleGraph V) (d : ℕ) : Prop :=
    68 Nonempty (OrientedPartitionSequence G d)

    71the graph admits an oriented `d`-sequence. -/
