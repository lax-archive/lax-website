      Natural Language Statement

      Definition

      The oracle decodes an LP certificate and either rejects the relaxation, returns a separating projection cut, or invokes the near-relaxation oracle. A bounded rounded-ellipsoid search repeats this step.

      The correctness predicates describe the approximation guarantee and the protected region that every returned cut must preserve. A failed search certifies that the independence number is below the requested target.

      38def tolerance (a K B n D : ℕ) : ℚ := momentFloor a K B n D / (32 * (n : ℚ)^2 * (D : ℚ)^2)
      39
      40def raw {N : ℕ} (center : Fin N → ℚ) : Fin N → ℚ := fun i => 4 * (N : ℚ) * center i
      41
      42def run {n : ℕ} (lp flow : List ℕ → List ℕ) (a K B D : ℕ)
      43 (G : SimpleGraph (Fin n)) [DecidableRel G.Adj] (k : ℕ)
      44 (child : Finset (Fin n) → ℕ → Option (Finset (Fin n)))
      45 (center : Index n (level a K B n D) → ℚ) :
      46 Reply (dimension n (level a K B n D)) (Finset (Fin n)) :=
      47 let P := program G k (level a K B n D)
      48 match classify P (tolerance a K B n D)
      49 (certificate lp (distanceProgram P (raw center))) with
      50 | .reject => .reject
      51 | .far normal _ => .cut (integerNormal normal)
      52 | .near y =>
      53 match NearOracle.run flow a K B D G k (decode y) child with
      54 | .reject => .reject
      55 | .found I => .found I
      56 | .cut A C u => .cut (integerNormal
      57 (rationalNormal (encode (coefficient A C u))))
      58
      59def Good {n : ℕ} (G : SimpleGraph (Fin n)) (k h D : ℕ) (I : Finset (Fin n)) : Prop :=
      60 G.IsIndepSet (I : Set (Fin n)) ∧ (1 - 4 * (1 / (D : ℝ))) ^ (h+1) * (k : ℝ) ≤ I.card
      61
      62def Correct {n : ℕ} (G : SimpleGraph (Fin n)) (k h a K B D : ℕ)
      63 (center : Index n (level a K B n D) → ℚ) :
      64 Reply (dimension n (level a K B n D)) (Finset (Fin n)) → Prop
      65 | .found I => Good G k h D I
      66 | .reject => independentSetHull G k (level a K B n D) = ∅
      67 | .cut normal => (∃ i, normal i ≠ 0) ∧
      68 ∀ z ∈ independentSetHull G k (level a K B n D),
      69 ∀ e : Index n (level a K B n D) → ℝ,
      70 (∀ i, |e i| ≤ ((2 : ℝ) ^ precision a K B n D)⁻¹) →
      71 (fun i => (normal i : ℝ)) ⬝ᵥ
      72 (scale (encode z) + e - (fun i => (center i : ℝ))) ≤ 0
      73
      74def search {n : ℕ} (lp flow : List ℕ → List ℕ) (a K B D : ℕ)
      75 (G : SimpleGraph (Fin n)) [DecidableRel G.Adj] (k : ℕ)
      76 (child : Finset (Fin n) → ℕ → Option (Finset (Fin n))) : Option (Finset (Fin n)) :=
      77 RoundedOracleSearch.search (fun _ center => run lp flow a K B D G k child center)
      78 (precision a K B n D)
      79
      80def SearchCorrect {n : ℕ} (G : SimpleGraph (Fin n)) (k h D : ℕ) : Option (Finset (Fin n)) → Prop
      81 | some I => Good G k h D I
      82 | none => IsIndependenceBound G Finset.univ (k-1)
