      19def HasPowerBagBound (C : GraphClass) (a A : ℕ) : Prop :=
      20 ∀ n (G : SimpleGraph (Fin n)), G ∈ C n → ∃ D : TreeDecomposition G,
      21 ∀ i, (D.bag i).card ^ a ≤ A * (G.edgeSet.ncard + 1) ^ (a - 1)
      22
      23def bagBound (a A m : ℕ) : ℕ := Nat.nthRoot a (A * (m + 1) ^ (a - 1))
      24
      25axiom root_bag_bound (C : GraphClass) (a A : ℕ) (a_positive : 0 < a) (A_positive : 0 < A)
      26 (class_bound : HasPowerBagBound C a A) :
      27 Monotone (bagBound a A) ∧ HasBagBound C (bagBound a A) ∧
      28 (∀ m, 1 ≤ bagBound a A m) ∧
      29 (∀ m, (bagBound a A m) ^ a ≤ A * (m + 1) ^ (a - 1))
