      Definition

      A thin symmetric loopless quiver has at most one arrow between each ordered pair of vertices, arrows exist symmetrically in both directions, and there are no arrows from a vertex to itself. Two such quivers on the same vertices are isomorphic here when their arrow types are equivalent for every ordered pair.

      28/-- A quiver has no parallel arrows. -/
      29def IsThin {V : Type u} (Q : Quiver.{v} V) : Prop :=
      30 ∀ u v : V, Subsingleton (Q.Hom u v)
      31
      32/-- A quiver has an arrow in one direction exactly when it has one in the reverse direction. -/
      33def IsSymmetric {V : Type u} (Q : Quiver.{v} V) : Prop :=
      34 ∀ u v : V, Nonempty (Q.Hom u v) ↔ Nonempty (Q.Hom v u)
      35
      36/-- A quiver has no arrows from a vertex to itself. -/
      37def IsLoopless {V : Type u} (Q : Quiver.{v} V) : Prop :=
      38 ∀ u : V, IsEmpty (Q.Hom u u)
      39
      40/-- A bundled quiver that is thin, symmetric, and loopless. -/
      41structure ThinSymmetricLooplessQuiver (V : Type u) : Type max (u + 1) (v + 1) where
      42 /-- The underlying quiver. -/
      43 quiver : Quiver.{v} V
      44 /-- There is at most one arrow with each fixed source and target. -/
      45 thin : IsThin quiver
      46 /-- Arrows exist symmetrically in the two directions. -/
      47 symmetric : IsSymmetric quiver
      48 /-- No arrow has equal source and target. -/
      49 loopless : IsLoopless quiver
      50
      51/-- Two quivers on the same vertices are isomorphic when all corresponding arrow types are equivalent. -/
      52structure Isomorphic {V : Type u} (Q : ThinSymmetricLooplessQuiver.{u, v} V)
      53 (R : ThinSymmetricLooplessQuiver.{u, w} V) where
      54 /-- The equivalence between arrows with each fixed source and target. -/
      55 homEquiv : ∀ u v : V, Q.quiver.Hom u v ≃ R.quiver.Hom u v
      56
      57/-- The quiver whose arrow types are lifts of a simple graph's adjacency propositions. -/
      58def ofSimpleGraph {V : Type u} (G : SimpleGraph V) :
      59 ThinSymmetricLooplessQuiver.{u, 0} V where
      60 quiver := ⟨fun u v ↦ PLift (G.Adj u v)⟩

      62 cases a

      Formalization notes

Hom-wise equivalence is the appropriate isomorphism notion because a quiver has no composition or additional structure to preserve.
