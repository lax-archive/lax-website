      Concept map

      Definition

      A batch consists of a finite set of vertices and a proposed teaching set for each vertex. It is admissible at radius one when the teaching sets are positive and uniformly bounded, separate every pair inside the batch, and each batch vertex certifies itself against every distinct concept outside the batch.

      A finite schedule records, for each vertex, the last batch containing it. Retaining the corresponding teaching sets defines its final assignment.

      9each vertex. It is admissible at radius one when the teaching sets are
      10positive and uniformly bounded, separate every pair inside the batch, and
      11each batch vertex certifies itself against every distinct concept outside the
      12batch.
      13
      14A finite schedule records, for each vertex, the last batch containing it.
      15Retaining the corresponding teaching sets defines its final assignment.
      16-/

      18namespace Lax16.BatchFramework
      19
      20open Lax16.TeachingMaps
      21
      22universe u
      23
      24/-- A finite batch of vertices together with proposed teaching sets. -/
      25structure BatchAssignment {V : Type u} (G : SimpleGraph V) where
      26 /-- The vertices assigned in this batch. -/
      27 vertices : Finset V
      28 /-- A total map whose values on `vertices` are the batch assignments. -/
      29 teaching : TeachingMap V 1
      30
      31/-- The three admissibility conditions from the batch construction. -/
      32def IsAdmissible {V : Type u} (G : SimpleGraph V) (d : ℕ)
      33 (B : BatchAssignment G) : Prop :=
      34 (∀ ⦃v x : V⦄, v ∈ B.vertices → x ∈ B.teaching v →
      35 x ∈ closedBall G 1 v) ∧
      36 (∀ v : V, v ∈ B.vertices → (B.teaching v).card ≤ d) ∧
      37 (∀ ⦃v w : V⦄, v ∈ B.vertices → w ∈ B.vertices →
      38 DistinctConcepts G 1 v w → Separates G 1 B.teaching v w) ∧
      39 (∀ ⦃v x : V⦄, v ∈ B.vertices → x ∉ B.vertices →
      40 DistinctConcepts G 1 v x →
      41 ∃ y ∈ B.teaching v, y ∉ closedBall G 1 x)
      42
      43/--
      44A finite batch schedule with an explicit, verified last batch for each
      45vertex. Carrying the last index as data keeps the final assignment readable.
      46-/
      47structure BatchSchedule {V : Type u} (G : SimpleGraph V) where
      48 /-- Number of batches in the schedule. -/
      49 batchCount : ℕ
      50 /-- The batches, in chronological order. -/
      51 batch : Fin batchCount → BatchAssignment G
      52 /-- The last batch containing each vertex. -/
      53 last : V → Fin batchCount
      54 /-- Every vertex belongs to its selected last batch. -/
      55 mem_last : ∀ v : V, v ∈ (batch (last v)).vertices
      56 /-- No later batch contains the vertex. -/
      57 last_is_last :
      58 ∀ (v : V) (i : Fin batchCount), v ∈ (batch i).vertices → i ≤ last v
      59
      60/-- The final map obtained by retaining each vertex's last assignment. -/
      61def BatchSchedule.finalTeaching {V : Type u} {G : SimpleGraph V}
      62 (S : BatchSchedule G) : TeachingMap V 1 :=
      63 fun v => (S.batch (S.last v)).teaching v
      64
      65end Lax16.BatchFramework
