      Concept map

      Definition

      A vertex vvv of a finite simple graph represents its closed ball of radius kkk: the vertices reachable from vvv by a walk of length at most kkk.

      A teaching map assigns a finite set of labels to every vertex. It is no-clash when every two distinct radius-kkk balls are distinguished by a label used by at least one of their teaching sets. It is positive when every label assigned to vvv lies in the ball represented by vvv.

      The ordinary and positive no-clash teaching dimensions are the least uniform widths of such maps. All subsequent results in this submission specialize these definitions to k=1k = 1k=1, where the represented balls are precisely the closed neighborhoods of the draft.

      1import Mathlib.Combinatorics.SimpleGraph.Paths
      2import Mathlib.Data.Nat.Lattice
      3
      4/-!
      5---
      6title: No-clash teaching for closed graph balls
      7type: definition
      8---
      9A vertex `v` of a finite simple graph represents its closed ball of radius
      10`k`: the vertices reachable from `v` by a walk of length at most `k`.
      11
      12A teaching map assigns a finite set of labels to every vertex. It is
      13no-clash when every two distinct radius-`k` balls are distinguished by a
      14label used by at least one of their teaching sets. It is positive when every
      15label assigned to `v` lies in the ball represented by `v`.
      16
      17The ordinary and positive no-clash teaching dimensions are the least uniform
      18widths of such maps. All subsequent results in this submission specialize
      19these definitions to `k = 1`, where the represented balls are precisely the
      20closed neighborhoods of the draft.
      21-/
      22
      23namespace Lax16.TeachingMaps
      24
      25universe u
      26
      27/-- The closed radius-`k` ball around `v`, expressed by bounded walk length. -/
      28def closedBall {V : Type u} (G : SimpleGraph V) (k : ℕ) (v : V) : Set V :=
      29 {w | ∃ p : G.Walk v w, p.length ≤ k}
      30
      31/-- The closed radius-`k` ball as a finset when the vertex type is finite. -/
      32noncomputable def closedBallFinset {V : Type u} [Fintype V]
      33 (G : SimpleGraph V) (k : ℕ) (v : V) : Finset V := by
      34 classical
      35 exact Finset.univ.filter fun w => w ∈ closedBall G k v
      36
      37/--
      38A radius-`k` teaching map assigns a finite set of vertex labels to each
      39radius-`k` concept. The index makes the represented neighborhood size
      40explicit in every later interface.
      41-/
      42abbrev TeachingMap (V : Type u) (_k : ℕ) := V → Finset V
      43
      44/-- Two vertices represent distinct concepts at radius `k`. -/
      45def DistinctConcepts {V : Type u} (G : SimpleGraph V) (k : ℕ)
      46 (v w : V) : Prop :=
      47 closedBall G k v ≠ closedBall G k w
      48
      49/-- A label witnesses one of the two sides of the symmetric difference. -/
      50def IsWitness {V : Type u} (G : SimpleGraph V) (k : ℕ)
      51 (x v w : V) : Prop :=
      52 x ∈ closedBall G k v ∧ x ∉ closedBall G k w
      53
      54/-- The teaching sets of `v` and `w` contain a label on which their concepts differ. -/
      55def Separates {V : Type u} (G : SimpleGraph V) (k : ℕ)
      56 (T : TeachingMap V k) (v w : V) : Prop :=
      57 ∃ x : V, (x ∈ T v ∨ x ∈ T w) ∧
      58 (IsWitness G k x v w ∨ IsWitness G k x w v)
      59
      60/-- Every pair of distinct concepts is separated. -/
      61def IsNoClash {V : Type u} (G : SimpleGraph V) (k : ℕ)
      62 (T : TeachingMap V k) : Prop :=
      63 ∀ ⦃v w : V⦄, DistinctConcepts G k v w → Separates G k T v w
      64
      65/-- Every teaching label is positive for the concept receiving it. -/
      66def IsPositive {V : Type u} (G : SimpleGraph V) (k : ℕ)
      67 (T : TeachingMap V k) : Prop :=
      68 ∀ ⦃v x : V⦄, x ∈ T v → x ∈ closedBall G k v
      69
      70/-- Every teaching set has cardinality at most `d`. -/
      71def HasWidthAtMost {V : Type u} {k : ℕ}
      72 (T : TeachingMap V k) (d : ℕ) : Prop :=
      73 ∀ v : V, (T v).card ≤ d
      74
      75/-- Every teaching set is nonempty. -/
      76def HasNonemptySets {V : Type u} {k : ℕ} (T : TeachingMap V k) : Prop :=
      77 ∀ v : V, (T v).Nonempty
      78
      79/-- There is a radius-`k` no-clash teaching map of width at most `d`. -/
      80def HasNCTDAtMost {V : Type u} (G : SimpleGraph V) (k d : ℕ) : Prop :=
      81 ∃ T : TeachingMap V k, IsNoClash G k T ∧ HasWidthAtMost T d
      82
      83/-- There is a positive radius-`k` no-clash teaching map of width at most `d`. -/
      84def HasPositiveNCTDAtMost {V : Type u} (G : SimpleGraph V)
      85 (k d : ℕ) : Prop :=
      86 ∃ T : TeachingMap V k,
      87 IsPositive G k T ∧ IsNoClash G k T ∧ HasWidthAtMost T d
      88
      89/-- The no-clash teaching dimension for closed balls of radius `k`. -/
      90noncomputable def nctd {V : Type u} (G : SimpleGraph V) (k : ℕ) : ℕ :=
      91 sInf {d : ℕ | HasNCTDAtMost G k d}
      92
      93/-- The positive no-clash teaching dimension for closed balls of radius `k`. -/
      94noncomputable def positiveNCTD {V : Type u}
      95 (G : SimpleGraph V) (k : ℕ) : ℕ :=
      96 sInf {d : ℕ | HasPositiveNCTDAtMost G k d}
      97
      98/-- The radius-`k` ball of `v` is contained in that of `w`. -/
      99def Dominates {V : Type u} (G : SimpleGraph V) (k : ℕ)
      100 (v w : V) : Prop :=
      101 closedBall G k v ⊆ closedBall G k w
      102
      103/-- Replace the teaching set at `target` by the set at `source`. -/
      104noncomputable def copyTeachingSet {V : Type u} {k : ℕ} (T : TeachingMap V k)
      105 (source target : V) : TeachingMap V k := by
      106 classical
      107 exact fun v => if v = target then T source else T v
      108
      109end Lax16.TeachingMaps
