import LaxDemo.EvenAddition

namespace LaxDemoProofs

theorem even_add {m n : Nat} :
    LaxDemo.EvenAddition.Even m →
    LaxDemo.EvenAddition.Even n →
    LaxDemo.EvenAddition.Even (m + n) := by
  rintro ⟨a, rfl⟩ ⟨b, rfl⟩
  refine ⟨a + b, ?_⟩
  simp only [LaxDemo.EvenAddition.twice]
  ac_rfl

end LaxDemoProofs
