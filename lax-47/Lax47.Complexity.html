Executable graph encodings and triangle-free approximation

Definition

Graphs are finite Boolean adjacency matrices with symmetry and looplessness certificates. Their machine word is the vertex count followed by the complete row-major matrix, with Booleans encoded by 000 and 111. An approximation is a function certified polynomial-time by the Lax51 finite-Turing model; its returned independent set is decoded from that certified function.

28/-- An executable simple graph on the labeled vertex set Fin⁡(n)\operatorname{Fin}(n)Fin(n). -/
29structure GraphCode (n : ℕ) where
30 adjacent : Fin n → Fin n → Bool
31 loopless : ∀ vertex, adjacent vertex vertex = false
32 symmetric : ∀ left right, adjacent left right = adjacent right left
33
34/-- The mathematical simple graph represented by a Boolean adjacency matrix. -/
35def GraphCode.graph {n : ℕ} (code : GraphCode n) : SimpleGraph (Fin n) where
36 Adj left right := code.adjacent left right = true
37 symm left right h := by
38 change code.adjacent right left = true
39 rw [← code.symmetric]
40 exact h
41 loopless := ⟨fun vertex ↦ by
42 simp [code.loopless]⟩
43
44@[simp] lemma GraphCode.graph_adj {n : ℕ} (code : GraphCode n)
45 (left right : Fin n) :
46 code.graph.Adj left right ↔ code.adjacent left right = true :=
47 Iff.rfl
48
49/-- Vertex count followed by the row-major adjacency matrix. -/
50def GraphCode.bits {n : ℕ} (code : GraphCode n) : BitString :=
51 n :: List.ofFn fun rank : Fin (n * n) ↦
52 let vertex := finProdFinEquiv.symm rank
53 bitWord (code.adjacent vertex.1 vertex.2)
54
55lemma GraphCode.bits_length {n : ℕ} (code : GraphCode n) :
56 code.bits.length = 1 + n * n := by
57 simp [GraphCode.bits, Nat.add_comm]
58
59/-- The edgeless executable graph. -/
60def GraphCode.empty (n : ℕ) : GraphCode n where
61 adjacent := fun _ _ ↦ false
62 loopless := by simp
63 symmetric := by simp

65/-- Decode the first nnn output bits as a vertex set. -/
66def decodeVertexSet (n : ℕ) (bits : BitString) : Finset (Fin n) :=
67 Finset.univ.filter fun vertex ↦ bits[vertex.1]? = some 1

69/-- A polynomial-time executable approximation for triangle-free Max Independent Set. -/
70structure TriangleFreeMISApproximation (ε : ℝ) where
71 program : PolytimeProgram
72 correctness : ∀ (n : ℕ) (code : GraphCode n),
73 code.graph.CliqueFree 3 →
74 let set := decodeVertexSet n (program.output code.bits)
75 code.graph.IsIndepSet set ∧
76 (code.graph.indepNum : ℝ) ≤
77 Real.rpow n ((1 : ℝ) / 2 - ε) * set.card
78
79/-- The vertex set decoded from the certified machine's output. -/
80def TriangleFreeMISApproximation.output
81 {ε : ℝ} (algorithm : TriangleFreeMISApproximation ε)
82 {n : ℕ} (code : GraphCode n) : Finset (Fin n) :=
83 decodeVertexSet n (algorithm.program.output code.bits)
84
85/-- The decoded output is independent on every triangle-free input. -/
86theorem TriangleFreeMISApproximation.independent
87 {ε : ℝ} (algorithm : TriangleFreeMISApproximation ε)
88 {n : ℕ} (code : GraphCode n) (triangleFree : code.graph.CliqueFree 3) :
89 code.graph.IsIndepSet (algorithm.output code) :=
90 (algorithm.correctness n code triangleFree).1
91
92/-- The decoded output has the claimed approximation ratio. -/
93theorem TriangleFreeMISApproximation.approximation
94 {ε : ℝ} (algorithm : TriangleFreeMISApproximation ε)
95 {n : ℕ} (code : GraphCode n) (triangleFree : code.graph.CliqueFree 3) :
96 (code.graph.indepNum : ℝ) ≤
97 Real.rpow n ((1 : ℝ) / 2 - ε) * (algorithm.output code).card :=
98 (algorithm.correctness n code triangleFree).2
