Lax47.Machine

An operational machine model for polynomial-time computation

concepts/Lax47/Machine.lean · lax-47

Definition

Computations are finite multi-stack Turing machines over binary input and output alphabets. A run is evaluated by the recursive bounded interpreter below. One unit of time is one transition of the fixed finite machine, and a polynomial-time program must prove that this interpreter reaches a halted configuration within c(n+1)kc(n+1)^kc(n+1)k transitions on every input of length nnn.

In particular, programs do not contain arbitrary Lean functions describing their answers, and time bounds are not annotations detached from evaluation. The semantic output function used below extracts the result returned by that bounded interpreter. Randomized algorithms receive a finite list of uniformly random input bits.

25/-- A finite binary input, output, certificate, or random seed. -/
26abbrev BitString := List Bool
27
28/-- A finite machine whose designated input and output alphabets are binary. -/
29abbrev BitMachine := Turing.TM2ComputableAux Bool Bool
30
31/--
32Run at most the supplied number of transitions, returning the first halted
33configuration and returning no result if the fuel is exhausted first.
34-/
35def runConfigWithin (tm : Turing.FinTM2) :
36 (fuel : ℕ) → tm.Cfg → Option tm.Cfg
37 | 0, cfg =>
38 match cfg.l with
39 | none => some cfg
40 | some _ => none
41 | fuel + 1, cfg =>
42 match cfg.l with
43 | none => some cfg
44 | some _ =>
45 match tm.step cfg with
46 | none => none
47 | some next => runConfigWithin tm fuel next
48
49/-- The binary output produced by a machine within the supplied fuel. -/
50def evalWithin (machine : BitMachine) (fuel : ℕ)
51 (input : BitString) : Option BitString :=
52 match runConfigWithin machine.tm fuel
53 (Turing.initList machine.tm
54 (input.map machine.inputAlphabet.invFun)) with
55 | none => none
56 | some cfg => some ((cfg.stk machine.tm.k₁).map machine.outputAlphabet)
57
58/-- The convenient monomial polynomial bound $c(n+1)^k$. -/
59def polynomialBound (c k n : ℕ) : ℕ :=
60 c * (n + 1) ^ k
61
62/-- A binary Turing program whose actual bounded evaluation always halts. -/
63structure PolytimeProgram where
64 machine : BitMachine
65 timeConstant : ℕ
66 timeExponent : ℕ
67 timeConstant_pos : 0 < timeConstant
68 terminates : ∀ input : BitString,
69 ∃ output, evalWithin machine
70 (polynomialBound timeConstant timeExponent input.length) input = some output
71
72/-- The certified bounded evaluation returns a result. -/
73theorem PolytimeProgram.output_isSome (program : PolytimeProgram)
74 (input : BitString) :
75 (evalWithin program.machine
76 (polynomialBound program.timeConstant program.timeExponent input.length)
77 input).isSome :=
78 Option.isSome_iff_exists.2 (program.terminates input)
79
80/-- The executable output obtained from the certified bounded machine run. -/
81def PolytimeProgram.output (program : PolytimeProgram)
82 (input : BitString) : BitString :=
83 (evalWithin program.machine
84 (polynomialBound program.timeConstant program.timeExponent input.length)
85 input).get (program.output_isSome input)
86
87/-- The extracted output is produced by the operational evaluator. -/
88theorem PolytimeProgram.output_spec (program : PolytimeProgram)
89 (input : BitString) :
90 evalWithin program.machine
91 (polynomialBound program.timeConstant program.timeExponent input.length) input =
92 some (program.output input) := by
93 simpa only [PolytimeProgram.output] using
94 Option.eq_some_of_isSome (program.output_isSome input)

96/-- A self-delimiting pairing of two binary strings. -/
97def pairBits (left right : BitString) : BitString :=
98 List.replicate left.length true ++ false :: left ++ right
99
100/-- A uniformly random string of exactly $r$ bits. -/
101abbrev RandomSeed (r : ℕ) := Fin r → Bool
102
103/-- The list representation of a fixed-length random seed. -/
104def RandomSeed.bits {r : ℕ} (seed : RandomSeed r) : BitString :=
105 List.ofFn seed
106
107/-- A decision problem over finite binary inputs. -/
108abbrev Language := Set BitString
109
110/-- A polynomial-time verifier with polynomially bounded certificates. -/
111structure NPVerifier (language : Language) where
112 program : PolytimeProgram
113 certificateConstant : ℕ
114 certificateExponent : ℕ
115 certificateConstant_pos : 0 < certificateConstant
116 correctness : ∀ input : BitString,
117 input ∈ language ↔ ∃ certificate : BitString,
118 certificate.length ≤ polynomialBound
119 certificateConstant certificateExponent input.length ∧
120 program.output (pairBits input certificate) = [true]
121
122/-- Membership in $NP$ in the operational binary-machine model. -/
123def InNP (language : Language) : Prop :=
124 Nonempty (NPVerifier language)
125
126/-- A polynomial-time randomized decision program using finitely many bits. -/
127structure BPPAlgorithm (language : Language) where
128 program : PolytimeProgram
129 randomnessConstant : ℕ
130 randomnessExponent : ℕ
131 randomnessConstant_pos : 0 < randomnessConstant
132 correctness : ∀ input : BitString,
133 let randomBitCount := polynomialBound
134 randomnessConstant randomnessExponent input.length
135 let seeds : Finset (RandomSeed randomBitCount) := Finset.univ
136 let accepting := seeds.filter fun seed ↦
137 program.output (pairBits input seed.bits) = [true]
138 (input ∈ language → 2 * seeds.card ≤ 3 * accepting.card) ∧
139 (input ∉ language → 3 * accepting.card ≤ seeds.card)
140
141/-- Membership in $BPP$ in the same operational binary-machine model. -/
142def InBPP (language : Language) : Prop :=
143 Nonempty (BPPAlgorithm language)
144
145/-- The complexity-class inclusion appearing in the inapproximability theorem. -/
146def NPSubsetBPP : Prop :=
147 ∀ language : Language, InNP language → InBPP language
