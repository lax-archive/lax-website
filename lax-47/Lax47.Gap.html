Definition

The Håstad premise is stated for executable promise-gap programs. A program is either a binary Turing program or the fixed triangle-removal pipeline used in this submission. The latter constructor is transparent: it runs the finite-bit reduction, invokes the supplied Turing approximation program on the encoded output graph, and compares two natural powers using the counted unary arithmetic below.

Thus acceptance, random-bit use, and running time are functions of the same execution. There is no semantic answer function and no unrelated step annotation. For an integer q>2q>2q>2, the high promise is n1−1/q≤α(H)n^{1-1/q}\leq\alpha(H)n1−1/q≤α(H), the low promise is α(H)≤n1/q\alpha(H)\leq n^{1/q}α(H)≤n1/q, and the acceptance threshold is nq+3≤∣S∣qn^{q+3}\leq |S|^qnq+3≤∣S∣q.

29/-- Length of the fixed unary-size adjacency-matrix graph encoding. -/
30def graphEncodingLength (n : ℕ) : ℕ :=
31 n + 1 + n * n
32
33/- ### Counted unary arithmetic used by the acceptance test -/
34
35/-- Unary multiplication paired with the work performed by its recursion. -/
36def countedMul (left : ℕ) : ℕ → ℕ × ℕ
37 | 0 => (0, 1)
38 | right + 1 =>
39 let prior := countedMul left right
40 (prior.1 + left, prior.2 + left + 1)
41
42/-- Unary exponentiation paired with the work performed by its recursion. -/
43def countedPow (base : ℕ) : ℕ → ℕ × ℕ
44 | 0 => (1, 1)
45 | exponent + 1 =>
46 let prior := countedPow base exponent
47 let product := countedMul prior.1 base
48 (product.1, prior.2 + product.2 + 1)
49
50/-- The concrete threshold comparison and its unary-operation count. -/
51def gapDecision (q n outputCard : ℕ) : Bool × ℕ :=
52 let left := countedPow n (q + 3)
53 let right := countedPow outputCard q
54 (decide (left.1 ≤ right.1),
55 left.2 + right.2 + left.1 + right.1 + 1)

59/--
60Codes accepted by the promise-gap model. The second constructor is a fixed
61composition primitive whose evaluator is defined below.
62-/
63inductive GapProgram (q : ℕ) where
64 | machine (program : PolytimeProgram)
65 (randomnessConstant randomnessExponent : ℕ)
66 | triangleReduction (ε : ℝ) (algorithm : TriangleFreeMISApproximation ε)

68/-- Finite uniform seed type read by a gap program on an $n$-vertex graph. -/
69def GapProgram.Seed {q : ℕ} : GapProgram q → ℕ → Type
70 | .machine _ constant exponent, n =>
71 RandomSeed (polynomialBound constant exponent (graphEncodingLength n))
72 | .triangleReduction _ _, n => ExecutionSeed n

86/-- Number of independent fair bits in the structured seed. -/
87def GapProgram.randomBitCount {q : ℕ} : GapProgram q → ℕ → ℕ
88 | .machine _ constant exponent, n =>
89 polynomialBound constant exponent (graphEncodingLength n)
90 | .triangleReduction _ _, n =>
91 Fintype.card (EdgeVariable n) * (executionBudget n + 1) *
92 executionSampleBits n
93
94/-- List representation of a machine-constructor seed. -/
95def machineSeedBits {r : ℕ} (seed : RandomSeed r) : BitString :=
96 List.ofFn seed
97
98/-- Boolean answer returned by the operational program. -/
99def GapProgram.accepts {q : ℕ} :
100 (program : GapProgram q) → (n : ℕ) → GraphCode n →
101 program.Seed n → Bool
102 | .machine program _ _, _, input, seed =>
103 decide (program.output
104 (pairBits input.bits (machineSeedBits seed)) = [true])
105 | .triangleReduction _ algorithm, n, input, seed =>
106 (gapDecision q n
107 (algorithm.output (executionOutput input seed)).card).1
108
109/-- Work charged to the same evaluation that returns $GapProgram.accepts$. -/
110def GapProgram.steps {q : ℕ} :
111 (program : GapProgram q) → (n : ℕ) → GraphCode n →
112 program.Seed n → ℕ
113 | .machine program _ _, _, input, seed =>
114 polynomialBound program.timeConstant program.timeExponent
115 (pairBits input.bits (machineSeedBits seed)).length
116 | .triangleReduction _ algorithm, n, input, seed =>
117 let reduced := executionOutput input seed
118 let output := algorithm.output reduced
119 executionSteps input seed +
120 polynomialBound algorithm.program.timeConstant
121 algorithm.program.timeExponent reduced.bits.length +
122 (n * n + 1) ^ 2 +
123 (gapDecision q n output.card).2
124
125/-- Uniform seed enumeration for an operational program. -/
126def GapProgram.seeds {q : ℕ} (program : GapProgram q) (n : ℕ) :
127 Finset (program.Seed n) :=
128 Finset.univ
129
130/-- Seeds on which the operational program returns true. -/
131def GapProgram.acceptingSeeds {q : ℕ} (program : GapProgram q)
132 (n : ℕ) (input : GraphCode n) : Finset (program.Seed n) :=
133 (program.seeds n).filter fun seed ↦ program.accepts n input seed

135/- ### Certified polynomial gap solvers -/
136
137/--
138A bounded-error polynomial-time solver for Håstad's rational promise gap.
139All resource functions are fixed by $GapProgram$; this structure supplies only
140their polynomial bounds and the two correctness proofs.
141-/
142structure MISGapSolver (q : ℕ) where
143 program : GapProgram q
144 timeConstant : ℕ
145 timeExponent : ℕ
146 timeConstant_pos : 0 < timeConstant
147 timeBound : ∀ (n : ℕ) (input : GraphCode n) (seed : program.Seed n),
148 program.steps n input seed ≤ polynomialBound timeConstant timeExponent n
149 randomnessConstant : ℕ
150 randomnessExponent : ℕ
151 randomnessConstant_pos : 0 < randomnessConstant
152 randomnessBound : ∀ n : ℕ, program.randomBitCount n ≤
153 polynomialBound randomnessConstant randomnessExponent n
154 cutoff : ℕ
155 completeness : ∀ (n : ℕ), cutoff ≤ n → ∀ input : GraphCode n,
156 Real.rpow n (1 - (q : ℝ)⁻¹) ≤ (input.graph.indepNum : ℝ) →
157 2 * (program.seeds n).card ≤
158 3 * (program.acceptingSeeds n input).card
159 soundness : ∀ (n : ℕ), cutoff ≤ n → ∀ input : GraphCode n,
160 (input.graph.indepNum : ℝ) ≤ Real.rpow n ((q : ℝ)⁻¹) →
161 3 * (program.acceptingSeeds n input).card ≤
162 (program.seeds n).card
