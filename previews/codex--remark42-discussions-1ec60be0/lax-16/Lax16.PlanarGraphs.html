      Concept map

      Definition

      A topological model of a graph HHH in a graph GGG maps the vertices of HHH injectively to branch vertices of GGG and maps every edge of HHH to a path in GGG. Path interiors contain no branch vertices and interiors belonging to distinct edges are disjoint.

      For finite simple graphs, planarity is encoded by Kuratowski's characterization: neither K5K₅K5​ nor K3,3K₃,₃K3​,3​ has a topological model. Outerplanarity is encoded by the analogous characterization excluding topological models of K4K₄K4​ and K2,3K₂,₃K2​,3​.

      7title: Planar and outerplanar finite simple graphs
      8type: definition
      9---
      10A topological model of a graph `H` in a graph `G` maps the vertices of `H`
      11injectively to branch vertices of `G` and maps every edge of `H` to a path in
      12`G`. Path interiors contain no branch vertices and interiors belonging to
      13distinct edges are disjoint.
      14
      15For finite simple graphs, planarity is encoded by Kuratowski's
      16characterization: neither `K₅` nor `K₃,₃` has a topological model.
      17Outerplanarity is encoded by the analogous characterization excluding
      18topological models of `K₄` and `K₂,₃`.
      19-/

      25/-- The internal vertices of a walk, with both endpoints removed. -/
      26def walkInterior {V : Type u} {G : SimpleGraph V} {a b : V}
      27 (p : G.Walk a b) : Set V :=
      28 {x | x ∈ p.support ∧ x ≠ a ∧ x ≠ b}
      29
      30/--
      31A subdivision model of `H` in `G`: distinct branch vertices joined by
      32internally vertex-disjoint paths corresponding to the edges of `H`.
      33-/
      34structure TopologicalModel {W : Type u} {V : Type v}
      35 (H : SimpleGraph W) (G : SimpleGraph V) where
      36 /-- The injective map from vertices of `H` to branch vertices of `G`. -/
      37 branch : W ↪ V
      38 /-- The path representing each oriented edge of `H`. -/
      39 route : ∀ {a b : W}, H.Adj a b → G.Walk (branch a) (branch b)
      40 /-- Every route is a graph-theoretic path. -/
      41 route_isPath : ∀ {a b : W} (h : H.Adj a b), (route h).IsPath
      42 /-- No branch vertex occurs internally on an edge route. -/
      43 branch_avoids_interiors :
      44 ∀ {a b : W} (h : H.Adj a b) (w : W),
      45 branch w ∉ walkInterior (route h)
      46 /-- Routes representing distinct undirected edges have disjoint interiors. -/
      47 route_interiors_disjoint :
      48 ∀ {a b c d : W} (hab : H.Adj a b) (hcd : H.Adj c d),
      49 ¬ ((a = c ∧ b = d) ∨ (a = d ∧ b = c)) →
      50 Disjoint (walkInterior (route hab)) (walkInterior (route hcd))
      51
      52/-- `H` occurs as a subdivision in `G`. -/
      53def HasTopologicalModel {W : Type u} {V : Type v}
      54 (H : SimpleGraph W) (G : SimpleGraph V) : Prop :=
      55 Nonempty (TopologicalModel H G)
      56
      57/-- Planarity via exclusion of subdivisions of `K₅` and `K₃,₃`. -/
      58def IsPlanar {V : Type v} (G : SimpleGraph V) : Prop :=
      59 ¬ HasTopologicalModel (⊤ : SimpleGraph (Fin 5)) G ∧
      60 ¬ HasTopologicalModel
      61 (completeBipartiteGraph (Fin 3) (Fin 3)) G
      62
      63/-- Outerplanarity via exclusion of subdivisions of `K₄` and `K₂,₃`. -/
      64def IsOuterplanar {V : Type v} (G : SimpleGraph V) : Prop :=
      65 ¬ HasTopologicalModel (⊤ : SimpleGraph (Fin 4)) G ∧
      66 ¬ HasTopologicalModel
      67 (completeBipartiteGraph (Fin 2) (Fin 3)) G
      68
      69/--
      70A graph is two-connected when it has at least three vertices and remains
      71connected after deletion of any single vertex.
      72-/
      73def IsTwoConnected {V : Type v} (G : SimpleGraph V) : Prop :=
      74 3 ≤ Nat.card V ∧
      75 ∀ v : V, (G.induce {w : V | w ≠ v}).Connected
