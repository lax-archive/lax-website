      Definition

      A finite wheel consists of a cycle of at least three rim vertices together with one hub adjacent to every rim vertex, and has no other edges.

      2import Lax68.HalinGraphs

      5/-!
      6---
      7title: Wheels
      8type: definition
      9---
      10A finite wheel consists of a cycle of at least three rim vertices together
      11with one hub adjacent to every rim vertex, and has no other edges.
      12-/

      14set_option autoImplicit false

      16namespace Lax68.Wheels
      17
      18def CycleAdjacent {n : ℕ} (i j : Fin n) : Prop :=
      19 i.val + 1 = j.val ∨
      20 j.val + 1 = i.val ∨
      21 (i.val = 0 ∧ j.val + 1 = n) ∨
      22 (j.val = 0 ∧ i.val + 1 = n)
      23
      24def IsCycleOn {V : Type*}
      25 (R : SimpleGraph V) (S : Set V) : Prop :=
      26 (∀ ⦃u v⦄, R.Adj u v → u ∈ S ∧ v ∈ S) ∧
      27 ∃ n : ℕ,
      28 3 ≤ n ∧
      29 ∃ e : Fin n ≃ {v : V // v ∈ S},
      30 ∀ i j,
      31 R.Adj (e i).1 (e j).1 ↔ CycleAdjacent i j
      32
      33def HasWheelShape {V : Type*} (G : SimpleGraph V) : Prop :=
      34 ∃ hub : V,
      35 hub ∈ G.universalVerts ∧
      36 ∃ rim : SimpleGraph V,
      37 IsCycleOn rim {v | v ≠ hub} ∧
      38 ∀ ⦃u v⦄,
      39 u ≠ hub →
      40 v ≠ hub →
      41 (G.Adj u v ↔ rim.Adj u v)
      42
      43def IsWheel {V : Type*} (G : SimpleGraph V) : Prop :=
      44 HasWheelShape G
