Lax47.Reduction

Finite-bit reduction to triangle-free Max Independent Set

concepts/Lax47/Reduction.lean · lax-47

definition

Concept map

Definition

This concept defines the fixed randomized reduction used by the proof. An nnn-vertex graph is sent to a graph on the n×nn\times nn×n blow-up of its vertices. The program receives one finite block of fair bits for every table cell that a polynomially truncated resampling loop can read. It scans the finite list of ordered vertex triples, resamples the three edges of the first present triangle, and stops after 12(n+1)612(n+1)^612(n+1)6 rounds. If the final scan still finds a triangle, it emits the edgeless graph.

The counter state and its operation count are returned by the same structural recursion. The mathematical infinite table used in the probability proof is not an input to this executable reduction.

33/-- Vertices of the complete nnn-fold blow-up. -/
34abbrev BlowupVertex (n : ℕ) := Fin n × Fin n
35
36/-- Edge variables used by the resampling construction. -/
37abbrev EdgeVariable (n : ℕ) := Sym2 (BlowupVertex n)
38
39/-- The infinite table of Boolean samples read lazily by Moser--Tardos. -/
40abbrev SampleTable (n : ℕ) :=
41 ResamplingTable (fun _ : EdgeVariable n ↦ Bool)
42
43/- ### Fixed finite-bit execution of the reduction -/
44
45/-- Ordered triples scanned by the executable triangle-removal loop. -/
46abbrev ExecutionTriple (n : ℕ) :=
47 BlowupVertex n × BlowupVertex n × BlowupVertex n
48
49/-- First vertex of an ordered execution triple. -/
50def executionFirstVertex {n : ℕ} (triple : ExecutionTriple n) : BlowupVertex n :=
51 triple.1
52
53/-- Second vertex of an ordered execution triple. -/
54def executionSecondVertex {n : ℕ} (triple : ExecutionTriple n) : BlowupVertex n :=
55 triple.2.1
56
57/-- Third vertex of an ordered execution triple. -/
58def executionThirdVertex {n : ℕ} (triple : ExecutionTriple n) : BlowupVertex n :=
59 triple.2.2
60
61/-- The three unordered edge variables of an ordered triple. -/
62def executionTriangleVariables {n : ℕ} (triple : ExecutionTriple n) :
63 Finset (EdgeVariable n) :=
64 {s(executionFirstVertex triple, executionSecondVertex triple),
65 s(executionFirstVertex triple, executionThirdVertex triple),
66 s(executionSecondVertex triple, executionThirdVertex triple)}
67
68/-- Number of fair bits used to realize one Bernoulli edge sample. -/
69def executionSampleBits (n : ℕ) : ℕ :=
70 Nat.log2 (100 * (n + 1)) + 1
71
72/-- Maximum number of Moser--Tardos rounds executed. -/
73def executionBudget (n : ℕ) : ℕ :=
74 12 * (n + 1) ^ 6
75
76/--
77The actual random input: one finite fair-bit block for every edge and every
78row that the bounded loop can read.
79-/
80abbrev ExecutionSeed (n : ℕ) :=
81 EdgeVariable n → Fin (executionBudget n + 1) →
82 Fin (executionSampleBits n) → Bool
83
84/-- Decode one block of fair bits as a Bernoulli sample. -/
85def executionBlockValue {n : ℕ}
86 (block : Fin (executionSampleBits n) → Bool) : Bool :=
87 decide (∀ bit, block bit = true)
88
89/-- Read a sampled edge cell, returning false outside the finite prefix. -/
90def executionCell {n : ℕ} (seed : ExecutionSeed n)
91 (edge : EdgeVariable n) (row : ℕ) : Bool :=
92 if h : row < executionBudget n + 1 then
93 executionBlockValue (seed edge ⟨row, h⟩)
94 else false
95
96/-- Boolean adjacency in the complete blow-up of an encoded input graph. -/
97def executionBlowupAdjacent {n : ℕ} (input : GraphCode n)
98 (left right : BlowupVertex n) : Bool :=
99 input.adjacent left.1 right.1
100
101/-- Whether an ordered triple is a currently present blow-up triangle. -/
102def executionViolates {n : ℕ} (input : GraphCode n)
103 (seed : ExecutionSeed n) (counts : EdgeVariable n → ℕ)
104 (triple : ExecutionTriple n) : Bool :=
105 executionBlowupAdjacent input (executionFirstVertex triple)
106 (executionSecondVertex triple) &&
107 executionBlowupAdjacent input (executionFirstVertex triple)
108 (executionThirdVertex triple) &&
109 executionBlowupAdjacent input (executionSecondVertex triple)
110 (executionThirdVertex triple) &&
111 executionCell seed s(executionFirstVertex triple, executionSecondVertex triple)
112 (counts s(executionFirstVertex triple, executionSecondVertex triple)) &&
113 executionCell seed s(executionFirstVertex triple, executionThirdVertex triple)
114 (counts s(executionFirstVertex triple, executionThirdVertex triple)) &&
115 executionCell seed s(executionSecondVertex triple, executionThirdVertex triple)
116 (counts s(executionSecondVertex triple, executionThirdVertex triple))
117
118/-- A conservative elementary-operation charge for testing one triple. -/
119def executionTestSteps (n : ℕ) : ℕ :=
120 20 * (executionBudget n + 1) * (executionSampleBits n + 1)
121
122/--
123Scan a concrete list of ordered triples. The returned step count is produced
124by the same recursion as the selected triple.
125-/
126def scanExecutionTriples {n : ℕ} (input : GraphCode n)
127 (seed : ExecutionSeed n) (counts : EdgeVariable n → ℕ) :
128 List (ExecutionTriple n) → Option (ExecutionTriple n) × ℕ
129 | [] => (none, 1)
130 | triple :: rest =>
131 if executionViolates input seed counts triple then
132 (some triple, executionTestSteps n)
133 else
134 let tail := scanExecutionTriples input seed counts rest
135 (tail.1, executionTestSteps n + tail.2)
136
137/-- Row-major rank of one blow-up vertex. -/
138def executionVertexEquiv (n : ℕ) : BlowupVertex n ≃ Fin (n * n) :=
139 finProdFinEquiv
140
141/--
142Mixed-radix rank of an ordered triple of blow-up vertices. Its value is
143u(n2)2+v(n2)+wu(n^2)^2+v(n^2)+wu(n2)2+v(n2)+w, where u,v,wu,v,wu,v,w are the row-major vertex ranks.
144-/
145def executionTripleEquiv (n : ℕ) :
146 ExecutionTriple n ≃ Fin ((n * n) * (n * n) * (n * n)) :=
147 let rankVertices := Equiv.prodCongr (executionVertexEquiv n)
148 (Equiv.prodCongr (executionVertexEquiv n) (executionVertexEquiv n))
149 let rankTail := Equiv.prodCongr (Equiv.refl (Fin (n * n)))
150 (finProdFinEquiv : Fin (n * n) × Fin (n * n) ≃
151 Fin ((n * n) * (n * n)))
152 let rankAll :=
153 (finProdFinEquiv : Fin (n * n) × Fin ((n * n) * (n * n)) ≃
154 Fin ((n * n) * ((n * n) * (n * n))))
155 rankVertices.trans <| rankTail.trans <| rankAll.trans <| finCongr (by ring)
156
157/-- Scan all n6n^6n6 ordered triples in mixed-radix order. -/
158def executionTriples (n : ℕ) : List (ExecutionTriple n) :=
159 List.ofFn fun rank : Fin ((n * n) * (n * n) * (n * n)) ↦
160 (executionTripleEquiv n).symm rank
161
162@[simp] lemma mem_executionTriples {n : ℕ} (triple : ExecutionTriple n) :
163 triple ∈ executionTriples n := by
164 rw [executionTriples, List.mem_ofFn]
165 exact ⟨executionTripleEquiv n triple,
166 (executionTripleEquiv n).symm_apply_apply triple⟩
167
168@[simp] lemma executionTriples_length (n : ℕ) :
169 (executionTriples n).length = (n * n) * (n * n) * (n * n) := by
170 simp [executionTriples]
171
172@[simp] lemma executionTriples_getElem {n i : ℕ}
173 (hi : i < (n * n) * (n * n) * (n * n)) :
174 (executionTriples n)[i]'(by simpa using hi) =
175 (executionTripleEquiv n).symm ⟨i, hi⟩ := by
176 simp [executionTriples]

178/-- Scan all n6n^6n6 ordered triples in the fixed finite enumeration. -/
179def findExecutionViolation {n : ℕ} (input : GraphCode n)
180 (seed : ExecutionSeed n) (counts : EdgeVariable n → ℕ) :
181 Option (ExecutionTriple n) × ℕ :=
182 scanExecutionTriples input seed counts (executionTriples n)
183
184/-- Increment exactly the three counters of a selected triangle. -/
185def advanceExecutionCounts {n : ℕ} (counts : EdgeVariable n → ℕ)
186 (selected : Option (ExecutionTriple n)) : EdgeVariable n → ℕ :=
187 selected.elim counts fun triple edge ↦
188 if edge ∈ executionTriangleVariables triple then counts edge + 1 else counts edge
189
190/--
191Run a fixed number of rounds. Result and step count are paired by this
192structural recursion; there is no independently supplied cost function.
193-/
194def executeRounds {n : ℕ} (input : GraphCode n) (seed : ExecutionSeed n) :
195 (rounds : ℕ) → (EdgeVariable n → ℕ) → (EdgeVariable n → ℕ) × ℕ
196 | 0, counts => (counts, 1)
197 | rounds + 1, counts =>
198 let prior := executeRounds input seed rounds counts
199 let scan := findExecutionViolation input seed prior.1
200 (advanceExecutionCounts prior.1 scan.1, prior.2 + scan.2 + 1)
201
202/-- Counter state after the polynomially bounded resampling loop. -/
203def executionCounts {n : ℕ} (input : GraphCode n) (seed : ExecutionSeed n) :
204 EdgeVariable n → ℕ :=
205 (executeRounds input seed (executionBudget n) (fun _ ↦ 0)).1
206
207/-- The loop has halted when its final deterministic scan finds no triangle. -/
208def executionHalted {n : ℕ} (input : GraphCode n)
209 (seed : ExecutionSeed n) : Bool :=
210 (findExecutionViolation input seed (executionCounts input seed)).1.isNone
211
212/-- Pair encoding of the vertices of the n×nn\times nn×n blow-up. -/
213def decodeBlowupVertex {n : ℕ} (vertex : Fin (n * n)) : BlowupVertex n :=
214 finProdFinEquiv.symm vertex
215
216@[simp] lemma decodeBlowupVertex_encode {n : ℕ} (vertex : BlowupVertex n) :
217 decodeBlowupVertex (finProdFinEquiv vertex) = vertex :=
218 finProdFinEquiv.symm_apply_apply vertex
219
220/-- The graph emitted by the fixed finite-bit reduction. -/
221def executionOutput {n : ℕ} (input : GraphCode n)
222 (seed : ExecutionSeed n) : GraphCode (n * n) where
223 adjacent left right :=
224 if executionHalted input seed then
225 let u := decodeBlowupVertex left
226 let v := decodeBlowupVertex right
227 input.adjacent u.1 v.1 &&
228 executionCell seed s(u, v) (executionCounts input seed s(u, v))
229 else false
230 loopless := by
231 intro vertex
232 simp [decodeBlowupVertex, input.loopless]
233 symmetric := by
234 intro left right
235 by_cases h : executionHalted input seed
236 · simp only [h, if_true, decodeBlowupVertex]
237 rw [input.symmetric]
238 rw [Sym2.eq_swap]
239 · simp [h]
240
241/-- Actual counted work of the bounded reduction, including its final scan. -/
242def executionSteps {n : ℕ} (input : GraphCode n)
243 (seed : ExecutionSeed n) : ℕ :=
244 (executeRounds input seed (executionBudget n) (fun _ ↦ 0)).2 +
245 (findExecutionViolation input seed (executionCounts input seed)).2 +
246 (n + 1) ^ 4
