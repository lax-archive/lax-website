      Definition

      A graph is outerplanar here when it has a crossing-free straight-line drawing with every vertex on one circle, a compact certificate for having every vertex on the boundary of the outer face.

      Lean source view on GitHub

      1import Lax68.Planar
      2import Lax68.StraightLineDrawings
      3
      4/-!
      5---
      6title: Outerplanar graphs
      7type: definition
      8---
      9A graph is outerplanar here when it has a crossing-free straight-line drawing
      10with every vertex on one circle, a compact certificate for having every
      11vertex on the boundary of the outer face.
      12-/
      13
      14set_option autoImplicit false
      15
      16namespace Lax68.Outerplanar
      17
      18structure OuterplaneDrawing {V : Type*} (G : SimpleGraph V)
      19 extends StraightLineDrawings.StraightLineDrawing G where
      20 radius : ℝ
      21 radius_pos : 0 < radius
      22 onBoundary :
      23 ∀ v : V,
      24 let p := toStraightLineDrawing.point v
      25 p.1 ^ 2 + p.2 ^ 2 = radius ^ 2
      26
      27def IsOuterplanar {V : Type*} (G : SimpleGraph V) : Prop :=
      28 Nonempty (OuterplaneDrawing G)
      29
      30/-- Every outerplanar graph is planar. -/
      31axiom outerplanar_planar {V : Type*} {G : SimpleGraph V} :
      32 Lax68.Outerplanar.IsOuterplanar G →
      33 Lax68.Planar.IsPlanar G
      34
      35end Lax68.Outerplanar
