3/-!
4---
5title: The Moser–Tardos theorem
6type: theorem
7---
8Let finitely many mutually independent random variables determine finitely many
9measurable bad events. If numbers $x(A)\in(0,1)$ satisfy
10$$
11 \Pr[A] \le x(A)\prod_{B\in\Gamma(A)}(1-x(B)),
12$$
13where $\Gamma(A)$ consists of the other bad events sharing a variable with
14$A$, then an assignment avoiding every bad event exists. Moreover, for any
15measurable deterministic rule that selects a currently true bad event, the
16sequential resampling algorithm has expected number of resamplings of $A$ at
17most $x(A)/(1-x(A))$, and its expected total number of resamplings is at most
18the sum of these bounds.
19-/

42/-- A coordinate of the infinite resampling table. -/
43abbrev TableIndex := (i : I) × Nat

45/-- An infinite table of fresh samples, one row for each independent variable. -/
46abbrev ResamplingTable := (j : TableIndex (I := I)) → Value j.1

48/-- An assignment to the variables in a finite scope. -/
49abbrev LocalAssignment (s : Finset I) := (i : s) → Value i.1

51/-- The product law of an infinite resampling table. -/
52noncomputable def tableMeasure (μ : ∀ i, MeasureTheory.Measure (Value i)) :
53 MeasureTheory.Measure (ResamplingTable Value) :=
54 MeasureTheory.Measure.infinitePi fun j : TableIndex (I := I) ↦ μ j.1

56/-- The product law restricted to a finite collection of variables. -/
57noncomputable def localMeasure (μ : ∀ i, MeasureTheory.Measure (Value i))
58 (s : Finset I) : MeasureTheory.Measure (LocalAssignment Value s) :=
59 MeasureTheory.Measure.infinitePi fun i : s ↦ μ i.1

61/-- The probability of a bad event under the original product distribution. -/
62noncomputable def eventProbability
63 (μ : ∀ i, MeasureTheory.Measure (Value i))
64 (scope : E → Finset I)
65 (bad : ∀ e, Set (LocalAssignment Value (scope e))) (e : E) : ℝ≥0∞ :=
66 localMeasure Value μ (scope e) (bad e)

68/-- A complete assignment to the finite family of independent variables. -/
69abbrev Assignment [Fintype I] := ∀ i, Value i

71/-- Restriction of a complete assignment to an event's scope. -/
72def restrictAssignment (scope : E → Finset I) (a : Assignment Value) (e : E) :
73 LocalAssignment Value (scope e) :=

76/-- The bad event `e` is true in assignment `a`. -/
77def violates (scope : E → Finset I)

79 (a : Assignment Value) (e : E) : Prop :=

82/-- A measurable deterministic strategy for choosing a currently true bad event. -/
83structure SelectionRule (scope : E → Finset I)
84 (bad : ∀ e, Set (LocalAssignment Value (scope e))) where
85 choose : Assignment Value → Option E
86 measurable_fiber : ∀ o, MeasurableSet {a | choose a = o}
87 sound : ∀ ⦃a : Assignment Value⦄ ⦃e : E⦄,
88 choose a = some e → violates Value scope bad a e
89 complete : ∀ ⦃a : Assignment Value⦄,
90 choose a = none → ∀ e, ¬violates Value scope bad a e

92/-- The assignment exposed by a resampling table at the given row counters. -/
93def currentAssignment (table : ResamplingTable Value) (counts : I → Nat) :
94 Assignment Value :=
95 fun i ↦ table ⟨i, counts i⟩
96
97/-- Increment precisely the counters in the scope of the selected event. -/
98def advanceCounts (scope : E → Finset I) (counts : I → Nat)
99 (selected : Option E) : I → Nat :=
100 selected.elim counts fun e i ↦ if i ∈ scope e then counts i + 1 else counts i

102/-- The row counters after the first `n` iterations of the resampling algorithm. -/
103def runCounts (scope : E → Finset I)
104 (bad : ∀ e, Set (LocalAssignment Value (scope e)))
105 (rule : SelectionRule Value scope bad) (table : ResamplingTable Value) :
106 Nat → I → Nat :=
107 Nat.rec (fun _ ↦ 0) fun _ counts ↦
108 advanceCounts scope counts (rule.choose (currentAssignment Value table counts))

110/-- The event resampled at time `n`, or `none` once no bad event remains. -/
111def resamplingLog (scope : E → Finset I)
112 (bad : ∀ e, Set (LocalAssignment Value (scope e)))
113 (rule : SelectionRule Value scope bad) (table : ResamplingTable Value)
114 (n : Nat) : Option E :=
115 rule.choose (currentAssignment Value table
116 (runCounts Value scope bad rule table n))

118/-- The (possibly infinite) number of times event `root` is resampled. -/
119noncomputable def resamplingCount (scope : E → Finset I)
120 (bad : ∀ e, Set (LocalAssignment Value (scope e)))
121 (rule : SelectionRule Value scope bad) (table : ResamplingTable Value)
122 (root : E) : ℝ≥0∞ :=
123 ∑' t : Nat, Set.indicator
124 {t | resamplingLog Value scope bad rule table t = some root}
125 (fun _ ↦ (1 : ℝ≥0∞)) t

127/-- The expected number of resamplings of one event. -/
128noncomputable def expectedResamplings
129 (μ : ∀ i, MeasureTheory.Measure (Value i)) (scope : E → Finset I)
130 (bad : ∀ e, Set (LocalAssignment Value (scope e)))
131 (rule : SelectionRule Value scope bad) (root : E) : ℝ≥0∞ :=
132 ∫⁻ table, resamplingCount Value scope bad rule table root
133 ∂tableMeasure Value μ

135/-- The odds $x/(1-x)$. -/
136noncomputable def odds (x : E → NNReal) (a : E) : NNReal :=
137 x a / (1 - x a)

139/-- A factor in the asymmetric local-lemma hypothesis. -/
140def localLemmaFactor (scope : E → Finset I) (x : E → NNReal)
141 (a b : E) : NNReal :=
142 if b ≠ a ∧ ¬Disjoint (scope a) (scope b) then 1 - x b else 1

144/-- The right side $x(A)\prod_{B\in\Gamma(A)}(1-x(B))$. -/
145noncomputable def localLemmaBound (scope : E → Finset I)
146 (x : E → NNReal) (a : E) : NNReal :=
147 x a * ∏ b : E, localLemmaFactor scope x a b
148
149/-- The sum of the expected resampling counts of all bad events. -/
150noncomputable def expectedTotalResamplings
151 (μ : ∀ i, MeasureTheory.Measure (Value i)) (scope : E → Finset I)
152 (bad : ∀ e, Set (LocalAssignment Value (scope e)))
153 (rule : SelectionRule Value scope bad) : ℝ≥0∞ :=
154 ∑ e : E, expectedResamplings Value μ scope bad rule e

156/-- The constructive asymmetric Lovász local lemma of Moser and Tardos. -/
157axiom moser_tardos
158 (μ : ∀ i, MeasureTheory.Measure (Value i))
159 [∀ i, MeasureTheory.IsProbabilityMeasure (μ i)]
160 (scope : E → Finset I)
161 (bad : ∀ e, Set (LocalAssignment Value (scope e)))
162 (hbad : ∀ e, MeasurableSet (bad e))
163 (rule : SelectionRule Value scope bad) (x : E → NNReal)
164 (hx_pos : ∀ a, 0 < x a) (hx_lt_one : ∀ a, x a < 1)
165 (hprob : ∀ a, eventProbability Value μ scope bad a ≤
166 (localLemmaBound scope x a : ℝ≥0∞)) :
167 (∃ a : Assignment Value, ∀ e, ¬violates Value scope bad a e) ∧
168 (∀ root, expectedResamplings Value μ scope bad rule root ≤
169 (odds x root : ℝ≥0∞)) ∧
170 expectedTotalResamplings Value μ scope bad rule ≤
171 ∑ e : E, (odds x e : ℝ≥0∞)
