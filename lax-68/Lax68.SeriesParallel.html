      Definition

      A finite two-terminal series-parallel graph is built from a single terminal edge by series and parallel composition. The side conditions say that the composed graphs meet only at the intended terminals, and the final support condition excludes unused isolated vertices.

      6title: Series-parallel graphs
      7type: definition
      8---
      9A finite two-terminal series-parallel graph is built from a single terminal
      10edge by series and parallel composition. The side conditions say that the
      11composed graphs meet only at the intended terminals, and the final support
      12condition excludes unused isolated vertices.
      13-/

      17namespace Lax68.SeriesParallel
      18
      19def edgeGraph {V : Type*} (s t : V) : SimpleGraph V :=
      20 SimpleGraph.fromRel fun u v =>
      21 (u = s ∧ v = t) ∨
      22 (u = t ∧ v = s)
      23
      24inductive TwoTerminal {V : Type*} : SimpleGraph V → V → V → Prop
      25 | edge (s t : V) (hne : s ≠ t) :
      26 TwoTerminal (edgeGraph s t) s t
      27 | series
      28 {G H : SimpleGraph V}
      29 {s m t : V}
      30 (left : TwoTerminal G s m)
      31 (right : TwoTerminal H m t)
      32 (meet :
      33 ∀ v,
      34 v ∈ G.support →
      35 v ∈ H.support →
      36 v = m) :
      37 TwoTerminal (G ⊔ H) s t
      38 | parallel
      39 {G H : SimpleGraph V}
      40 {s t : V}
      41 (left : TwoTerminal G s t)
      42 (right : TwoTerminal H s t)
      43 (meet :
      44 ∀ v,
      45 v ∈ G.support →
      46 v ∈ H.support →
      47 v = s ∨ v = t) :
      48 TwoTerminal (G ⊔ H) s t
      49
      50def IsSeriesParallel {V : Type*} (G : SimpleGraph V) : Prop :=
      51 ∃ s t,
      52 TwoTerminal G s t ∧
      53 G.support = Set.univ
      54
      55/-- Every series-parallel graph is planar.
      56
      57Open in this formalization: no proof is supplied yet. -/
      58axiom seriesParallel_planar {V : Type*} {G : SimpleGraph V} :
      59 Lax68.SeriesParallel.IsSeriesParallel G →
      60 Lax68.Planar.IsPlanar G
